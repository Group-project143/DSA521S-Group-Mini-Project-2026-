// DSA compare sorting algorithms 
import java.util.Arrays;
import java.util.Random;

public class Main {

    static long comparisons = 0;

    //  Selection Sort 
    static void selectionSort(int[] arr) {
        int n = arr.length;
        for (int i = 0; i < n - 1; i++) {
            int minIdx = i;
            for (int j = i + 1; j < n; j++) {
                comparisons++;               // data comparison
                if (arr[j] < arr[minIdx]) {
                    minIdx = j;
                }
            }
            int temp = arr[minIdx];
            arr[minIdx] = arr[i];
            arr[i] = temp;
        }
    }

    // Insertion Sort 
    static void insertionSort(int[] arr) {
        int n = arr.length;
        for (int i = 1; i < n; i++) {
            int key = arr[i];
            int j = i - 1;
            while (j >= 0) {
                comparisons++;               // data comparison
                if (arr[j] > key) {
                    arr[j + 1] = arr[j];
                    j--;
                } else {
                    break;
                }
            }
            arr[j + 1] = key;
        }
    }

    // Merge Sort 
    static void mergeSort(int[] arr, int left, int right) {
        if (left < right) {
            int mid = (left + right) / 2;
            mergeSort(arr, left, mid);
            mergeSort(arr, mid + 1, right);
            merge(arr, left, mid, right);
        }
    }

    static void merge(int[] arr, int left, int mid, int right) {
        int n1 = mid - left + 1;
        int n2 = right - mid;
        int[] L = new int[n1];
        int[] R = new int[n2];
        System.arraycopy(arr, left, L, 0, n1);
        System.arraycopy(arr, mid + 1, R, 0, n2);

        int i = 0, j = 0, k = left;
        while (i < n1 && j < n2) {
            comparisons++;                   // data comparison
            if (L[i] <= R[j]) {
                arr[k++] = L[i++];
            } else {
                arr[k++] = R[j++];
            }
        }
        while (i < n1) arr[k++] = L[i++];
        while (j < n2) arr[k++] = R[j++];
    }

    //  Quick Sort 
    static void quickSort(int[] arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }

    static int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; j++) {
            comparisons++;                   // data comparison
            if (arr[j] < pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        return i + 1;
    }

    // runs one algorithm and print a result row
    interface SortCall {
        void run(int[] a);
    }

    static void runAndReport(String name, int size, int[] original, SortCall sorter) {
        int[] arr = Arrays.copyOf(original, original.length); // fresh copy, not timed
        comparisons = 0;

        long start = System.nanoTime();       // timing starts here
        sorter.run(arr);
        long end = System.nanoTime();          // timing ends here

        long elapsed = end - start;

        if (!isSorted(arr)) {
            System.out.println("ERROR: " + name + " did not sort correctly for size " + size);
        }

        System.out.printf("%-6d %-12s %-15d %-15d%n", size, name, comparisons, elapsed);
    }

    static boolean isSorted(int[] arr) {
        for (int i = 1; i < arr.length; i++) {
            if (arr[i - 1] > arr[i]) return false;
        }
        return true;
    }

    // Builds an almost-sorted array from a base array 
    static int[] makeAlmostSorted(int[] base) {
        int[] arr = Arrays.copyOf(base, base.length);
        Arrays.sort(arr); // fully sorted first (not timed/counted - just setup)

        int n = arr.length;
        int gap = n / 6; // 5 swaps spread evenly across the array
        for (int k = 1; k <= 5; k++) {
            int idx = Math.min(k * gap, n - 2); // index of first of the pair
            int temp = arr[idx];
            arr[idx] = arr[idx + 1];
            arr[idx + 1] = temp;
        }
        return arr;
    }

    public static void main(String[] args) {
        int[] sizes = {20, 50, 100, 500};
        Random rand = new Random(42); // fixed seed -> reproducible test arrays

        System.out.printf("%-6s %-12s %-15s %-15s%n", "Size", "Algorithm", "Comparisons", "Time (ns)");
        System.out.println("-------------------------------------------------------");

        int[] original100 = null; // keep the random size-100 array for the almost-sorted test

        for (int size : sizes) {
            // Generate ONE original array per size (not timed, not counted)
            int[] original = new int[size];
            for (int i = 0; i < size; i++) {
                original[i] = rand.nextInt(10000);
            }
            if (size == 100) {
                original100 = original;
            }

            // Each algorithm gets its own identical copy of 'original'
            runAndReport("Selection", size, original, Main::selectionSort);
            runAndReport("Insertion", size, original, Main::insertionSort);
            runAndReport("Merge",     size, original, a -> mergeSort(a, 0, a.length - 1));
            runAndReport("Quick",     size, original, a -> quickSort(a, 0, a.length - 1));

            System.out.println();
        }

        // Additional test: almost-sorted 100
        int[] almostSorted = makeAlmostSorted(original100);

        System.out.println("=== Almost-sorted array (n = 100, 5 neighbouring swaps) ===");
        System.out.printf("%-6s %-12s %-15s %-15s%n", "Size", "Algorithm", "Comparisons", "Time (ns)");
        System.out.println("-------------------------------------------------------");

        runAndReport("Selection", 100, almostSorted, Main::selectionSort);
        runAndReport("Insertion", 100, almostSorted, Main::insertionSort);
        runAndReport("Merge",     100, almostSorted, a -> mergeSort(a, 0, a.length - 1));
        runAndReport("Quick",     100, almostSorted, a -> quickSort(a, 0, a.length - 1));
    }
}
